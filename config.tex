% Dynamic display of the three roles depending on plurality and locale
\ExplSyntaxOn
\newcommand{\ajustProfessorName}{
    % Load list in a 'clist' variable (comma-separated list)
    \clist_set:No \l_tmpa_clist { \myprofessors }
    
    % Count the elements
    \int_compare:nNnTF { \clist_count:N \l_tmpa_clist } > { 1 }
    { % Plural case
        \ifdefstring{\reportlanguage}{french}
            {\renewcommand{\professorname}{Professeurs}}
            {\renewcommand{\professorname}{Professors}}
    }
    { % Singular case
        \ifdefstring{\reportlanguage}{french}
            {\renewcommand{\professorname}{Professeur}}
            {\renewcommand{\professorname}{Professor}}
    }
}
\ExplSyntaxOff
\ExplSyntaxOn
\newcommand{\ajustTeachingAssistantName}{
    % Load list in a 'clist' variable (comma-separated list)
    \clist_set:No \l_tmpa_clist { \myteachingassistant }
    
    % Count the elements
    \int_compare:nNnTF { \clist_count:N \l_tmpa_clist } > { 1 }
    { % Plural case
        \ifdefstring{\reportlanguage}{french}
            {\renewcommand{\teachingassistantname}{Assistants}}
            {\renewcommand{\teachingassistantname}{Teaching assistants}}
    }
    { % Singular case
        \ifdefstring{\reportlanguage}{french}
            {\renewcommand{\teachingassistantname}{Assistant}}
            {\renewcommand{\teachingassistantname}{Teaching assistant}}
    }
}
\ExplSyntaxOff
\ExplSyntaxOn
\newcommand{\ajustAuthorName}{
    % Load list in a 'clist' variable (comma-separated list)
    \clist_set:No \l_tmpa_clist { \myauthors }
    
    % Count the elements
    \int_compare:nNnTF { \clist_count:N \l_tmpa_clist } > { 1 }
    { % Plural case
        \ifdefstring{\reportlanguage}{french}
            {\renewcommand{\authorname}{Auteurs}}
            {\renewcommand{\authorname}{Authors}}
    }
    { % Singular case
        \ifdefstring{\reportlanguage}{french}
            {\renewcommand{\authorname}{Auteur}}
            {\renewcommand{\authorname}{Author}}
    }
}
\ExplSyntaxOff
